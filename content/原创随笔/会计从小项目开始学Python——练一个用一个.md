### 四、数据的三范式（3NF)

什么是数据的范式：是关系型数据库 / 结构化数据（比如 Excel 表格）的**数据组织规则**，核心目标是：
1. 消除数据冗余（比如客户名称不用在每张发票里都写一遍）；
2. 避免数据更新异常（改客户名称只需改一处）；
3. 保证数据一致性（财务数据核对不出现矛盾）。

三范式是逐级满足的：满足第二范式（2NF）必须先满足第一范式（1NF），满足第三范式（3NF）必须先满足第二范式（2NF）。

基础知识：一张表格，坚向排列的叫列，横向排列的叫行，交叉部分叫单元格。每列的第一行叫列标题或字段值，每行的第一列叫索引（如果设置有）。

第一范式（1NF）：列不可再分，每格只有一个值且保证原子性。即：
1. 同一列属性一致：如“部门”这一列，就不能填写关于“职务”的数据，
2. 保证原子性：如“电话号码”这一列，某一格的数据不能写成”13912345678，010-61234567“，它们没有保证”原子性“，如果需要同时记录，则分成两列，一列是”手机号码“，一列是”固定电话“。

第二范式：消除部分依赖，必须有主键，非主键列完全依赖主键。

第三范式：消除传递依赖，非主键列不依赖其他非主键列。

后两种范式，请大家自行问询 AI。提示词可以是：EXCEL 数据三范式等。

大家使用 EXCEL 时，太喜欢用单元格跨列居中这个功能了，它是数据清洗最大工作量之一，所以，在数据建表之初遵守三范式，可节省后期很多时间。

在使用 EXCEL时，有一个小技巧，一张数据表，使用三张表来分工合作：第一张表存放源数据（调整成满足三范式），第二张表是工作表，在这张表应用函数、辅助列等方法，完成工作任务，第三张表是呈现表，这里可以使用跨列居中等各种格式，制作出符合需要、美观的报表。这三张表之间使用数据引用的方式来保证三张表的同步。

### （一）合并一个 EXCEL 文件中的多个 sheet 表

场景：有一家连锁公司，每个月从收银软件导出营业流水放在了一个EXCEL文件中（收银软件的功能），每个店的数据存放在单独的 sheet 表中。会计每个月都需要将所有门店的数据汇总在一张 sheet 表中，方便统计。

合并多个 sheet 表，在 EXCEL 的 Power Query 可以实现，WPS 也有相应的功能。这里用 Python 练习它，是用一个最容易理解的工作任务来入门。

提示词：我有一个 EXCEL 文件，里面有多张 sheet 表，存放着相同数据结构的不同数据内容 ，我需要将这些 sheet 表，纵向汇集在一张 sheet 表中 ，且新增一列，填写每行数据来源于哪张表，要提供源文件和绝对路径的输入、sheet 表名的输入、合并后文件名和绝对路径的输入。请将上述内容，用 Python 编写代码并实现它。

AI 给出的代码：
```python
import pandas as pd
import os
from pathlib import Path

def merge_excel_sheets():
    """
    合并Excel多sheet页，支持：
    1. 自动移除路径中的中英文单/双引号
    2. 自动将sheet名分隔符的中文逗号转为英文逗号
    """

    print("===== Excel多sheet合并工具 =====")
    # 1. 输入源文件绝对路径
    while True:
        source_input = input("\n请输入需要合并的Excel文件绝对路径（含文件名，如：C:\\test\\源文件.xlsx）：").strip()
        # 移除所有中英文单/双引号
        source_input = source_input.replace('"', '').replace("'", '').replace('“', '').replace('”', '').replace('‘', '').replace('’', '')
        # 规范化路径
        try:
            source_file = Path(source_input).resolve()
        except:
            print("错误：路径格式无效，请重新输入！")
            continue
        source_file = str(source_file)
        break

    # 2. 输入需要合并的sheet名
    while True:
        sheet_input = input("\n请输入需要合并的sheet页名称（多个名称用逗号分隔，如：1店,2店,3店）：").strip()
        # 核心修复：将中文逗号替换为英文逗号，再拆分
        sheet_input = sheet_input.replace('，', ',')  # 中文逗号→英文逗号
        # 拆分+去空格+去空值+去重（避免重复sheet名）
        sheet_names = []
        for name in sheet_input.split(','):
            clean_name = name.strip()
            if clean_name and clean_name not in sheet_names:
                sheet_names.append(clean_name)
        break

    # 3. 接收输出文件绝对路径
    while True:
        output_input = input("\n请输入合并后文件的保存绝对路径（含文件名）：").strip()
        # 移除所有中英文单/双引号
        output_input = output_input.replace('"', '').replace("'", '').replace('“', '').replace('”', '').replace('‘', '').replace('’', '')
        try:
            output_file = Path(output_input).resolve()
        except:
            print("错误：路径格式无效，请重新输入！")
            continue
        output_file = str(output_file)
        break

    # 4. 核心合并逻辑
    try:
        print(f"\n正在读取文件：{source_file}")
        sheet_dict = pd.read_excel(source_file, sheet_name=sheet_names)
        df_list = []
        for SheetName, df in sheet_dict.items():
            df['表名'] = SheetName
            df_list.append(df)
            print(f"行数：{len(df_list)}")
        if df_list:
            big_df = pd.concat(df_list, ignore_index=True)
            big_df.to_excel(output_file, index=False)
            print(f"\n✅ 合并完成！")
            print(f"📁 保存路径：{output_file}")
            print(f"📊 总行数：{len(big_df)}")
        else:
            print("\n❌ 无有效数据可合并！")
    except Exception as e:
        print(f"\n❌ 出错：{str(e)}")

# 5. 调用自定义函数
if __name__ == "__main__":
    try:
        merge_excel_sheets()
    except Exception as e:
        print(f"程序运行出错：{str(e)}")
        # 兼容打包后stdin的问题，强制等待输入
        try:
            input("\n按回车键退出...")
        except:
            # 若仍无法读取stdin，延时退出，让用户看到报错
            import time
            time.sleep(10)
```

## （二）合并同一文件夹里具有相同格式的多个EXCEL文件

场景：这个场景更常用，大多信息管理软件导出的数据是按月存放的，合并它们也是一个极大的工作量。

提示词：我有一个文件夹，里面存放着多个 EXCEL 文件，它们具有相同的格式，我需要在一个新的 EXCEL 文件中，纵向合并它们。
按这个提示词，AI 可能不会一次给出满足要求的代码，需要我们反复去调整提示词。

AI 给出的代码：
```python
from pathlib import Path
import pandas as pd
import os

# ---------------------- 编写函数 ----------------------
def merge_excel_files(input_dir: Path, output_file: Path):
    """合并Excel文件核心逻辑"""
    all_data = []
    excel_files = list(input_dir.glob("*.xlsx")) + list(input_dir.glob("*.xls"))
    
    if not excel_files:
        print("⚠️  目标目录下未找到Excel文件！")
        return
    
    for file in excel_files:
        file_name = file.name
        print(f"正在处理文件：{file_name}")
        
        try:
            excel_dict = pd.read_excel(file, sheet_name=None, keep_default_na=False)
        except Exception as e:
            print(f"❌ 读取文件 {file_name} 失败：{e}")
            continue
        
        for sheet_name, df in excel_dict.items():
            print(f"  - 处理工作表：{sheet_name}")
            df["数据来源"] = f"{file_name}-{sheet_name}"
            all_data.append(df)
    
    if all_data:
        merged_df = pd.concat(all_data, axis=0, ignore_index=True)
        with pd.ExcelWriter(output_file, engine="openpyxl") as writer:
            merged_df.to_excel(writer, sheet_name="SHEET1", index=False)
        
        print(f"\n✅ 合并完成！结果已保存至：{output_file.absolute()}")
        print(f"📊 合并后总行数：{len(merged_df)}，总列数：{len(merged_df.columns)}")
    else:
        print("❌ 无有效数据可合并！")

# ---------------------- 主程序（修复盘符路径问题） ----------------------
if __name__ == "__main__":
    print("="*50)
    print("        Excel文件合并工具（最终版）")
    print("="*50)
    
    while True:
        # 1. 获取用户输入
        input_dir_str = input("\n请输入待合并Excel文件的文件夹路径：\n（示例：C:/Users/Administrator/Desktop/合并表格文件）\n").strip()
        
        # 2. 路径清洗：移除冒号 + 去引号 + 修复反斜杠
        cleaned_dir_str = input_dir_str.replace("：", "").replace(":", "")  # 去冒号
        cleaned_dir_str = cleaned_dir_str.strip('"').strip("'")            # 去引号
        cleaned_dir_str = cleaned_dir_str.replace("\\", "/")               # 反斜杠转斜杠
        
        # 3. 核心修复：补全盘符后的斜杠（解决C/Users变成相对路径的问题）
        if len(cleaned_dir_str) >= 2 and cleaned_dir_str[1] == "/" and cleaned_dir_str[0].isalpha():
            # 匹配 "C/Users..." 格式，修复为 "C:/Users..."
            cleaned_dir_str = f"{cleaned_dir_str[0]}:/{cleaned_dir_str[2:]}"
        
        # 4. 调试：打印处理后的路径
        print(f"\n🔍 调试信息 - 最终处理路径：{cleaned_dir_str}")
        
        # 5. 校验路径
        try:
            input_dir = Path(cleaned_dir_str).resolve()  # 转为绝对路径
            print(f"🔍 调试信息 - 解析后的绝对路径：{input_dir}")
            
            if not input_dir.exists():
                print(f"❌ 错误：路径「{input_dir}」不存在！请检查文件夹是否存在/路径拼写。")
                continue
            if not input_dir.is_dir():
                print(f"❌ 错误：「{input_dir}」不是文件夹！")
                continue
            if not os.access(input_dir, os.R_OK):
                print(f"❌ 错误：无访问「{input_dir}」的权限！")
                continue
            
            print("✅ 路径校验通过！")
            break
        
        except Exception as e:
            print(f"❌ 路径解析出错：{type(e).__name__} - {e}")
            # 兜底：弹出文件夹选择窗口（无需手动输路径）
            try:
                from tkinter import Tk
                from tkinter.filedialog import askdirectory
                Tk().withdraw()
                print("\n💡 自动打开文件夹选择窗口，请手动选择...")
                input_dir = Path(askdirectory(title="选择待合并Excel的文件夹"))
                if input_dir.is_dir():
                    print(f"✅ 手动选择成功：{input_dir}")
                    break
                else:
                    print("❌ 未选择有效文件夹！")
            except:
                print("❌ 无法打开选择窗口，请按示例格式输入路径！")
                continue
    
    # 构造输出文件
    output_file = input_dir / "合并文件.xlsx"
    
    # 确认执行
    print(f"\n📌 最终确认：")
    print(f"  - 待处理文件夹：{input_dir.absolute()}")
    print(f"  - 输出文件路径：{output_file.absolute()}")
    confirm = input("\n是否确认开始合并？(y/n)：").strip().lower()
    
    if confirm == "y":
        merge_excel_files(input_dir, output_file)
    else:
        print("🚫 操作已取消！")

```
